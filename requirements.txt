# Snake Pi — dependências (Raspberry Pi OS Bookworm, Python 3.11)
#
# No Bookworm o pip global é bloqueado ("externally-managed-environment").
# O caminho recomendado é instalar pelo apt (ver install.sh ou o README):
#
#   sudo apt install python3-pygame python3-smbus2 python3-spidev \
#                    python3-gpiozero python3-lgpio i2c-tools
#
# Este arquivo existe para quem preferir venv:
#   python3 -m venv --system-site-packages venv
#   source venv/bin/activate && pip install -r requirements.txt

pygame>=2.5.0          # renderização do jogo no monitor HDMI
gpiozero>=2.0          # abstração de GPIO (botões, buzzers, 74HC595)
lgpio>=0.2.2.0         # backend de GPIO recomendado no Bookworm/Pi 4
smbus2>=0.4.3          # I2C puro: ADS7830 e LCD1602
spidev>=3.6            # SPI: leitor RFID
mfrc522>=0.0.7         # driver de alto nível do RC522
